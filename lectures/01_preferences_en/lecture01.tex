\begin{frame}[label=why]{Why do we need microeconomics?}
\begin{enumerate}[<+->]
\item \key{Describe the world with explicit, reproducible models.} State assumptions, derive mechanisms and test predictions. Build structural models for real-world data and policy counterfactuals.
\item \key{Start from people and firms.} Build aggregate outcomes from their decisions. After the Lucas critique (1976), microfoundations became central to much of modern macroeconomics. Central banks and research institutes use these tools regularly.
\item \key{Make and challenge economic arguments in court.} In US antitrust and damages cases, economists write expert reports, build counterfactuals and scrutinise one another's evidence.
\end{enumerate}

\begin{frame}[label=choice-set]{What is the primitive?}
\key{The choice set $X$: fully specified alternatives.}
\begin{itemize}[<+->]
\item \key{Discrete:} $X=\{\text{basic},\text{standard},\text{pro}\}$.
\item \key{Continuous:} $X=\mathbb R_+^L$, quantities of $L$ commodities.
\item \key{Discrete--continuous:} tea with $m$ units of milk, or coffee with $m$ units of milk:
\[
X=\{\text{tea},\text{coffee}\}\times\mathbb R_+.

% Stable ID: weak-preference
\begin{frame}[label=weak-preference]{One relation is enough to start}
\uncover<1->{A weak preference relation is a subset $\succeq\,\subseteq X\times X$.}
\uncover<2->{\[
x\succeq y\quad\Longleftrightarrow\quad
\text{``$x$ is at least as good as $y$.''}
\]}
\uncover<3->{It is a statement about a pair of alternatives, not a numerical score or an observed purchase.}
\uncover<4->{Some introductory textbooks also introduce $\sim$ or $\succ$ separately. Do we need them as additional primitives?}
\end{frame}

% Stable ID: derive-indifference
\begin{frame}[label=derive-indifference]{Can we define indifference using only $\succeq$?}
\uncover<1->{We already know how to say ``$x$ is at least as good as $y$''.}
\uncover<2->{How could we say ``$x$ and $y$ are equally good''?}
\uncover<3->{\[
\boxed{x\sim y\iff x\succeq y\ \text{and}\ y\succeq x.}
\]}
\uncover<4->{\par Both comparisons hold. This differs from incomparability, where neither comparison holds.}
\end{frame}

% Stable ID: derive-strict
\begin{frame}[label=derive-strict]{Can we define strict preference using only $\succeq$?}
\uncover<1->{What extra condition makes ``at least as good'' become ``strictly better''?}
\uncover<2->{\[
\boxed{x\succ y\iff x\succeq y\ \text{and not }y\succeq x.}
\]}
\uncover<3->{\par A second independent preference primitive is unnecessary.}
\uncover<4->{\par Does ``not $x\succ y$'' imply ``$y\succ x$''?}
\uncover<5->{\par No: $x\sim y$ is already a counterexample. Without completeness, incomparability is another.}
\end{frame}

% Stable ID: rationality
\begin{frame}[label=rationality]{What does rationality mean in a model?}
\uncover<1->{Economists love to assume \key{rational preferences}. What consistency would a model actually need?}
\uncover<2->{\par\key{Completeness:} for every $x,y\in X$, $x\succeq y$ or $y\succeq x$, or both.}
\uncover<3->{\par\key{Transitivity:} for every $x,y,z\in X$,
\[
x\succeq y,\ y\succeq z\quad\Longrightarrow\quad x\succeq z.
\]}
\uncover<4->{\par\key{Rational preferences = completeness + transitivity.}}
\uncover<5->{\par This does not require selfishness, perfect information or easy computation.}
\end{frame}

% Stable ID: relations
\begin{frame}[label=relations]{Your turn: three relations}
\begin{enumerate}[<+->]
\item On $\mathbb R_+^2$: $x\succeq y\iff x_1+x_2\ge y_1+y_2$.
\item On $\mathbb R^3$: $x\succeq y$ iff $x_i\ge y_i$ in \key{at least two of the three coordinates}.
\item On $\mathbb R_+^2$: $x\succeq y\iff x_1\ge y_1$ and $x_2\ge y_2$.
\end{enumerate}
\uncover<4->{\par Check completeness and transitivity. Give a counterexample when a property fails.}
\uncover<5->{\par\small For relation 2, try $(3,2,1)$, $(2,1,3)$ and $(1,3,2)$.}
\end{frame}

% Stable ID: observed-choice
\begin{frame}[label=observed-choice]{We see choices, not the whole preference relation}
\uncover<1->{A theory of preferences is useful, but we do not observe everybody's complete ranking. Usually we see choices from particular menus.}
\uncover<2->{\[
C(\{A,B,C\})=\{A\},\qquad C(\{B,C\})=\{B\}.
\]}
\uncover<3->{These choices are consistent with $A\succ B\succ C$: the menu changed, while preferences need not have changed.}
\uncover<4->{\par What must we observe besides a purchase before inferring anything about preferences?}
\end{frame}

% Stable ID: choice-rule
\begin{frame}[label=choice-rule]{Choice as the primitive}
\uncover<+->{Let $\mathcal D=2^X\setminus\{\varnothing\}$ be the collection of all nonempty menus.}

\uncover<+->{\begin{block}{Choice correspondence}
\[
C:\mathcal D\to 2^X,\qquad \varnothing\ne C(A)\subseteq A.
\]
$C(A)$ records \key{all selected best alternatives}, not just one realised purchase.
\end{block}}

\uncover<+->{A preference relation induces}

\uncover<+->{\[
C(A;\succeq)=\{x\in A:x\succeq y\text{ for every }y\in A\}.
\]}

\uncover<+->{\small A best element need not exist on an infinite menu. Nonemptiness is an additional requirement.}
\end{frame}

% Stable ID: warp
\begin{frame}[label=warp]{Consistency across menus: WARP}
\uncover<+->{\begin{block}{Weak Axiom of Revealed Preference}
For any menus $A,B$ and any $x,y\in A\cap B$,
\[
\begin{gathered}
x\in C(A),\quad y\in C(B)\\[2pt]
\Rightarrow\quad x\in C(B),\quad y\in C(A).
\end{gathered}
\]
\end{block}}

\begin{itemize}
\item Both alternatives must be available in both menus.
\item One cannot be strictly beaten in one menu and selected over its rival in another.
\item This formulation concerns the \key{whole correspondence}, including ties.
\end{itemize}
\end{frame}

% Stable ID: warp-example
\begin{frame}[label=warp-example]{A two-menu inconsistency}
\uncover<+->{\begin{columns}[T]
\column{.47\textwidth}\begin{block}{Menu $A=\{a,b,c\}$}
\[
C(A)=\{a\}.
\]
$a$ is selected; $b$ is available but not selected.
\end{block}
\column{.47\textwidth}\begin{block}{Menu $B=\{a,b,d\}$}
\[
C(B)=\{b\}.
\]
$b$ is selected; $a$ is available but not selected.
\end{block}
\end{columns}}

\uncover<+->{\question{Can a single complete and transitive preference relation generate both choice sets?}}

\end{frame}

% Stable ID: equivalence
\begin{frame}[label=equivalence]{Preferences and choice: an equivalence}
\uncover<+->{\begin{block}{Finite-set theorem}
Let $X$ be finite and $C$ be nonempty on \key{every} nonempty subset of $X$.

Then $C$ satisfies WARP if and only if there exists a complete, transitive relation $\succeq$ such that
\[
C(A)=C(A;\succeq)\quad\text{for every nonempty }A\subseteq X.
\]
\end{block}}

\begin{itemize}
\item Preferences $\to$ choices: select every best alternative.
\item Choices $\to$ preferences: infer $x\succeq y$ from $x\in C(\{x,y\})$.
\end{itemize}

\uncover<+->{\navto{proof-equivalence}{Proof, if time permits}\qquad\navto{proof-warp}{Preferences $\to$ WARP}}
\end{frame}

% Stable ID: choice-data
\begin{frame}[label=choice-data]{What choice data do not automatically reveal}
\begin{enumerate}
\item We often observe one realised choice, not \key{all} best alternatives.
\item We observe only a sample of menus, not every subset of $X$.
\item A consistency test on that sample need not establish global rationalisability.
\end{enumerate}

\uncover<+->{\takeaway{Observations, tie-breaking assumptions and the menu domain jointly determine what can be inferred.}}

\uncover<+->{\small If we only know that an observed choice is optimal, universal indifference can rationalise any such observations. Additional structure matters.}
\end{frame}

% Stable ID: utility
\begin{frame}[label=utility]{Utility is a representation}
\uncover<+->{\begin{block}{Definition}
$u:X\to\mathbb R$ represents $\succeq$ if, for every $x,y\in X$,
\[
x\succeq y\quad\Longleftrightarrow\quad u(x)\ge u(y).
\]
\end{block}}

\uncover<+->{Then}

\uncover<+->{\[
C(A;\succeq)=\arg\max_{x\in A}u(x).
\]}

\begin{itemize}
\item The numbers reproduce the entire ranking, including indifference.
\item Representation turns choice into an optimisation problem.
\end{itemize}
\end{frame}

% Stable ID: finite-utility
\begin{frame}[label=finite-utility]{Representation on a finite set}
\uncover<+->{Complete and transitive preferences on a finite $X$ admit a utility representation.}

\begin{enumerate}
\item Partition alternatives into indifference classes.
\item Order the classes from worst to best.
\item Assign increasing numerical labels.
\end{enumerate}

\uncover<+->{\[
\underbrace{\{C,D\}}_{u=0}\ \prec\
\underbrace{\{B\}}_{u=1}\ \prec\
\underbrace{\{A,E\}}_{u=2}.
\]}

\uncover<+->{\takeaway{The ranking matters. The gaps between the labels do not.}}
\end{frame}

% Stable ID: transformations
\begin{frame}[label=transformations]{Increasing transformations preserve the ranking}
\uncover<1->{If $f$ is strictly increasing on $u(X)$, then $v=f\circ u$ represents the same preferences:
\[
u(x)\ge u(y)\iff f(u(x))\ge f(u(y)).
\]}
\uncover<2->{\par Examples: $u+5$, $30u$, $e^u$. Squaring works on a nonnegative range, not in general.}
\uncover<3->{\par Why is a \key{weakly} increasing transformation not enough?}
\end{frame}

% Stable ID: interpersonal
\begin{frame}[label=interpersonal]{No interpersonal comparisons for free}
\uncover<+->{Yaromir and Zoya both prefer $x$ to $y$.}

\begin{itemize}
\item We may rescale Yaromir's utility without changing any of his choices.
\item We may independently rescale Zoya's utility without changing her choices.
\item The sum $u_Y+u_Z$ can change its ranking of social outcomes after those rescalings.
\end{itemize}

\uncover<+->{\takeaway{An individual ordinal representation does not, by itself, supply interpersonal units or a social welfare criterion.}}
\end{frame}

% Stable ID: lexicographic
\begin{frame}[label=lexicographic]{Does every ranking have a utility representation?}
\uncover<+->{Can every complete and transitive preference relation be represented by a real-valued utility function?}

\uncover<+->{On $X=[0,1]^2$, compare the first coordinate first, and the second only to break ties:}

\uncover<+->{\[
x\succeq_{\mathrm{lex}}y\iff
[x_1>y_1]\ \text{or}\ [x_1=y_1\text{ and }x_2\ge y_2].
\]}

\begin{itemize}
\item Complete and transitive.
\item Any gain in coordinate 1 outweighs any loss in coordinate 2.
\item No real-valued utility function represents this relation on $[0,1]^2$.
\end{itemize}

\uncover<+->{\small The non-representation proof uses uncountably many disjoint intervals; see the appendix.}
\end{frame}

% Stable ID: continuity
\begin{frame}[label=continuity]{Continuity: comparisons survive limits}
\uncover<+->{On $X\subseteq\mathbb R^L$, preferences are continuous if}

\uncover<+->{\[
\begin{gathered}
x^n\to x,\quad y^n\to y,\quad x^n\succeq y^n\ \forall n\\[2pt]
\Rightarrow\quad x\succeq y.
\end{gathered}
\]}

\uncover<+->{\begin{block}{A useful representation theorem}
On $\mathbb R_+^L$, complete, transitive and continuous preferences admit a \key{continuous} utility representation.
\end{block}}

\uncover<+->{\small Lexicographic counterexample: $(1/n,0)\succ_{\mathrm{lex}}(0,1)$ for all $n$, but $(0,0)\prec_{\mathrm{lex}}(0,1)$ in the limit.}
\end{frame}

% Stable ID: monotonicity
\begin{frame}[label=monotonicity]{Monotonicity: fix the notation first}
\uncover<+->{Write $x\ge y$ if $x_i\ge y_i$ for every $i$, and $x\gg y$ if $x_i>y_i$ for every $i$.}

\uncover<+->{\begin{block}{Two useful conditions}
\key{Monotonicity:}\quad $x\gg y\Rightarrow x\succ y$.\\[1mm]
\key{Strong monotonicity:}\quad $x\ge y,\ x\ne y\Rightarrow x\succ y$.
\end{block}
\begin{itemize}
\item More of \key{every} good versus more of \key{at least one} good, with no losses.
\item Neither is part of the definition of rationality.
\item The assumption is inappropriate for a bad unless we redefine the commodity.
\end{itemize}}
\end{frame}

% Stable ID: lns
\begin{frame}[label=lns]{Local nonsatiation: a better bundle nearby}
\uncover<+->{\key{Coffee and cake:} could there be a bundle where every sufficiently small change makes you no better off?}

\uncover<+->{For every $x\in X$ and every $\varepsilon>0$, there is $y\in X$ such that}

\uncover<+->{\[
\|y-x\|<\varepsilon,\qquad y\succ x.
\]}

\begin{itemize}
\item There is no local maximum of any representing utility function.
\item On $\mathbb R_+^L$: strong monotonicity $\Rightarrow$ monotonicity $\Rightarrow$ LNS.
\item The converse implications need not hold.
\end{itemize}

\uncover<+->{\begin{block}{Example}
$u(x_1,x_2)=x_1$ is locally nonsatiated but not strongly monotone.
\end{block}}
\end{frame}

% Stable ID: convex
\begin{frame}[label=convex]{Convex preferences: mixing bundles}
\uncover<1->{Assume $X$ is convex. Preferences are \key{convex} if, for every $\lambda\in[0,1]$,
\[
x\succeq z,\ y\succeq z\quad\Rightarrow\quad
\lambda x+(1-\lambda)y\succeq z.
\]}
\uncover<2->{\par Equivalently, every upper contour set $\{x:x\succeq z\}$ is convex.}
\uncover<3->{\par Preferences are \key{strictly convex} if, for distinct $x,y$ and $0<\lambda<1$, the conclusion is strict: $\lambda x+(1-\lambda)y\succ z$.}
\uncover<4->{\par In particular, mixing two distinct indifferent bundles is weakly better under convexity and strictly better under strict convexity.}
\uncover<5->{\par This is a mixture of quantities, not a lottery over the two bundles.}
\end{frame}

% Stable ID: strict-convex
\begin{frame}[label=strict-convex]{Convex functions and convex preferences}
\uncover<1->{On a convex domain, $u$ is a \key{convex function} if
\[
u(\lambda x+(1-\lambda)y)\le\lambda u(x)+(1-\lambda)u(y),
\qquad\lambda\in[0,1].
\]}
\uncover<2->{\par Do convex preferences imply a convex utility function? Must a representing utility be convex?}
\uncover<3->{\par\key{First example:} $u(x)=x_1+x_2$. It is affine: both convex and concave. The preferences are convex.}
\uncover<4->{\par\key{Increasing transformation:} $v(x)=e^{x_1+x_2}$. It represents the same preferences; it is convex but not concave.}
\uncover<5->{\par These two examples do not establish the general claim. Which property of utility actually captures convex preferences?}
\end{frame}

% Stable ID: quasiconcavity
\begin{frame}[label=quasiconcavity]{Convex preferences and quasiconcavity}
\uncover<1->{A convex representation need not exist. For $u(x)=\sqrt{x_1x_2}$,
\[
a=(1,4)\sim b=(4,1),\qquad m=\tfrac12(a+b)=(2.5,2.5)\succ a.
\]}
\uncover<2->{\par Any convex representation $g$ would require
\[
g(m)\le\tfrac12g(a)+\tfrac12g(b)=g(a),
\]
contradicting $m\succ a$.}
\uncover<3->{\par The right property is \key{quasiconcavity}:
\[
u(\lambda x+(1-\lambda)y)\ge\min\{u(x),u(y)\}.
\]}
\uncover<4->{\par Convex preferences $\Longleftrightarrow$ quasiconcavity of a representing utility. This property survives strictly increasing transformations.}
\end{frame}

% Stable ID: separability
\begin{frame}[label=separability]{Separability: a ranking inside a larger bundle}
\uncover<+->{Let $X=Y\times Z$. Think of $Y$ as food and $Z$ as housing and transport.}

\uncover<+->{\begin{block}{Preferences over $Y$ do not depend on $Z$}
For all $y,y'\in Y$ and $z,z'\in Z$,
\[
(y,z)\succeq(y',z)\quad\Longleftrightarrow\quad
(y,z')\succeq(y',z').
\]
\end{block}}

\begin{itemize}
\item Your ranking of two food bundles does not depend on which car you have.
\item Your \key{food budget} can still depend on car spending.
\item The condition is not symmetric between $Y$ and $Z$.
\end{itemize}
\end{frame}

% Stable ID: separable-aggregator
\begin{frame}[label=separable-aggregator]{Separability: summarise a group of goods}

\uncover<1->{Let $y=(y_1,y_2)$ be a food bundle and $z$ everything else. Suppose the ranking of food bundles does not depend on $z$.}
\uncover<2->{\par Summarise the food bundle by one number $v(y)$, then combine it with $z$:
\[
u(y,z)=U(v(y),z).
\]}
\uncover<3->{\par For each fixed $z$, $U$ must increase strictly with the food index: a better food bundle must still be better.}
\uncover<4->{\par\key{Example:}
\[
v(y)=\sqrt{y_1y_2},\qquad U(v,z)=v+\log(1+z).
\]
$(1,4)$ has food index $2$; $(1,1)$ has food index $1$.}
\uncover<5->{\par At \key{any fixed $z$}, the first food bundle ranks higher. This does not fix the food budget.\quad\navto{proof-separability}{Formal construction}}
\end{frame}

% Stable ID: quasilinearity
\begin{frame}[label=quasilinearity]{Quasilinearity: a numeraire plus a valuation}
\uncover<+->{\[
u(a,y)=a+v(y),\qquad a\ge0.
\]}

\begin{itemize}
\item $a$ is money left over, or another \key{numeraire} good.
\item $v(y)$ measures the valuation of $y$ in numeraire units.
\item Adding the same feasible amount $t$ to both numeraire holdings preserves their ranking.
\end{itemize}

\uncover<+->{\[
(a,y)\succeq(a',y')\iff(a+t,y)\succeq(a'+t,y').
\]}

\uncover<+->{\takeaway{A convenient assumption with economic content: money compensation is independent of the starting numeraire level.}}
\end{frame}

\begin{frame}[label=proof-warp]{Appendix: why rational preferences imply WARP}
\uncover<+->{Suppose $x,y\in A\cap B$, $x\in C(A)$ and $y\in C(B)$.}

\begin{enumerate}
\item Because $y\in A$, optimality of $x$ gives $x\succeq y$.
\item Because $x\in B$, optimality of $y$ gives $y\succeq x$.
\item For every $z\in B$, $y\succeq z$; transitivity gives $x\succeq z$.
\item Therefore $x\in C(B)$. The symmetric argument gives $y\in C(A)$.
\end{enumerate}

\uncover<+->{\takeaway{The argument uses transitivity and the full best-alternative interpretation of $C$. Finite menus ensure nonemptiness separately.}}
\vfill\navto{equivalence}{Back to the lecture}
\end{frame}

\begin{frame}[label=proof-equivalence]{Appendix: reconstructing the relation from choice}
\small
\uncover<+->{Define $x\succeq y$ iff $x\in C(\{x,y\})$.}

\begin{itemize}
\item \key{Completeness:} pair menus have nonempty chosen sets.
\item \key{Transitivity:} let $x\succeq y\succeq z$ and choose $w\in C(\{x,y,z\})$. WARP with the relevant pair menus first brings $y$ or $x$ into the triple, and then brings $x$ into it. WARP with $\{x,z\}$ gives $x\succeq z$.
\item \key{Representation of choice:} if $x\in C(A)$, WARP with each $\{x,y\}$ yields $x\succeq y$. Conversely, if $x$ beats every $y\in A$, choose $y\in C(A)$ and use WARP with $\{x,y\}$ to obtain $x\in C(A)$.
\end{itemize}

\uncover<+->{\small Domain: all nonempty subsets of a finite $X$. Repeated alternatives in a triple give trivial cases.}
\vfill\navto{equivalence}{Back to the lecture}
\end{frame}

\begin{frame}[label=proof-limited-data]{Appendix: limited data can hide a cycle}
\uncover<+->{Suppose only pair menus are observed:}

\uncover<+->{\[
C(\{s,p\})=\{s\},\qquad C(\{p,r\})=\{p\},\qquad C(\{s,r\})=\{r\}.
\]}

\begin{itemize}
\item Different observed menus share only one alternative.
\item WARP has no nontrivial two-alternative comparison across these menus.
\item Yet the singleton chosen sets require $s\succ p\succ r\succ s$.
\item No complete, transitive preference relation generates these full chosen sets.
\end{itemize}

\uncover<+->{\takeaway{Passing WARP on an arbitrary observed domain does not imply rationalisability on the full domain.}}
\vfill\navto{choice-data}{Back to the lecture}
\end{frame}

\begin{frame}[label=proof-lexicographic]{Appendix: why lexicographic utility does not exist}
\uncover<+->{Suppose a real-valued $u$ represents lexicographic preferences on $[0,1]^2$.}

\begin{enumerate}
\item For each $t\in[0,1]$, $u(t,1)>u(t,0)$.
\item Thus $I_t=(u(t,0),u(t,1))$ is a nonempty open interval.
\item If $s>t$, then $u(s,0)>u(t,1)$, so $I_s$ and $I_t$ are disjoint.
\item Each $I_t$ contains a rational number; distinct intervals contain distinct selected rationals.
\end{enumerate}

\uncover<+->{There cannot be an injection from uncountably many $t\in[0,1]$ into the countable rationals. Contradiction.}
\vfill\navto{lexicographic}{Back to the lecture}
\end{frame}

\begin{frame}[label=proof-continuity]{Appendix: construction under monotonicity}
\small
\uncover<+->{Assume rationality, continuity and monotonicity on $\mathbb R_+^L$; let $e=(1,\ldots,1)$.}

\begin{enumerate}
\item Continuity and monotonicity imply $x\ge y\Rightarrow x\succeq y$.
\item On $[0,M]$, $M=\max_i x_i$, the sets $\{a:ae\succeq x\}$ and $\{a:x\succeq ae\}$ are nonempty, closed and cover the interval. Connectedness makes them intersect.
\item If $a>b$, $ae\succ be$; hence the indifferent coefficient is unique. Set $u(x)=a$.
\item Transitivity and diagonal monotonicity give $x\succeq y\iff u(x)\ge u(y)$.
\item If $x^n\to x$, the coefficients $u(x^n)$ are bounded. Every convergent subsequence has limit $u(x)$ by continuity and uniqueness. Therefore $u(x^n)\to u(x)$.
\end{enumerate}
\vfill\navto{diagonal}{Back to the lecture}
\end{frame}

\begin{frame}[label=proof-separability]{Appendix: separability in one construction}
\small
\uncover<+->{Fix $z_0$ and let $v(y)=u(y,z_0)$. For $t\in v(Y)$, pick $\hat y(t)$ with $v(\hat y(t))=t$, and define}

\uncover<+->{\[
U(t,z)=u(\hat y(t),z).
\]}

\begin{itemize}
\item If two representatives have the same $t$, they are indifferent at $z_0$; independence makes them indifferent at every $z$. So $U$ is well-defined.
\item Since $\hat y(v(y))$ and $y$ are indifferent at every $z$,
\[
U(v(y),z)=u(y,z).
\]
\item If $t>t'$, strict preference at $z_0$ carries to every $z$, so $U(t,z)>U(t',z)$.
\end{itemize}

\uncover<+->{\small The constructed aggregator has domain $v(Y)\times Z$.}
\vfill\navto{separable-aggregator}{Back to the lecture}
\end{frame}

\begin{frame}[label=proof-quasilinearity]{Appendix: primitives behind quasilinearity}
\footnotesize
\uncover<+->{On $\mathbb R_+\times Y$, assume rationality, a worst $\bar y$, and:}

\begin{enumerate}
\item $(a,\bar y)\succeq(a',\bar y)\iff a\ge a'$.
\item For every $y$, there exists $t\ge0$ with $(0,y)\sim(t,\bar y)$.
\item Adding the same feasible amount to both numeraire holdings preserves comparisons in both directions.
\end{enumerate}

\uncover<+->{Define $v(y)$ by $(0,y)\sim(v(y),\bar y)$. Translation invariance gives}

\uncover<+->{\[
(a,y)\sim(a+v(y),\bar y).
\]}

\uncover<+->{Transitivity and condition 1 then imply}

\uncover<+->{\[
(a,y)\succeq(a',y')\iff a+v(y)\ge a'+v(y').
\]}

\uncover<+->{So $u(a,y)=a+v(y)$ represents preferences.}
\vfill\navto{quasilinearity}{Back to the lecture}

\end{frame}

\begin{frame}[label=proof-budget]{Appendix: LNS and budget exhaustion}
\small
\uncover<+->{Suppose $p\gg0$, $x^*$ is optimal and preferences are locally nonsatiated.}

\uncover<+->{If $p\cdot x^*<m$, let $\delta=m-p\cdot x^*>0$.}

\uncover<+->{Whenever $y\in X$ satisfies $\|y-x^*\|<\delta/\|p\|$,}

\uncover<+->{\[
p\cdot y\le p\cdot x^*+\|p\|\|y-x^*\|<m.
\]}

\uncover<+->{LNS gives a strictly better $y$ within this neighbourhood. It is feasible, contradicting optimality.}

\uncover<+->{\takeaway{LNS implies budget exhaustion at an existing optimum; it does not establish existence.}}
\vfill\navto{why-lns}{Back to the lecture}
\end{frame}

\begin{frame}[label=proof-risk]{Appendix: increasing transformations and risk}
\uncover<+->{Compare $A=2$ for sure with $B=0$ or $3$, each with probability $1/2$.}

\uncover<+->{\[
\mathbb E[A]=2>1.5=\mathbb E[B].
\]}

\uncover<+->{After the strictly increasing transformation $f(t)=e^t$,}

\uncover<+->{\[
\mathbb E[e^A]=e^2\approx7.39<\frac{1+e^3}{2}\approx10.54=\mathbb E[e^B].
\]}

\uncover<+->{\takeaway{The ranking of individual outcomes is unchanged, but the ranking of lotteries by expected transformed numbers changes. Ordinal deterministic utility and expected utility require different care.}}
\vfill\navto{transformations}{Back to the lecture}
\end{frame}
