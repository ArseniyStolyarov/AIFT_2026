\begin{frame}[label=consumer-problem]{The consumer problem}
\[
\begin{aligned}
\uncover<1->{\max_x}\quad&\uncover<1->{u(x)}\\[1em]
\uncover<2->{\text{subject to}}\quad&\uncover<2->{p\cdot x\le m,}\\[.7em]
&\uncover<3->{x_i\ge0\quad\text{for every }i.}
\end{aligned}
\]
\uncover<4->{Here $p\gg0$ and $m>0$. The feasible set is\[B(p,m)=\{x\ge0:p\cdot x\le m\}.\]}\par
\end{frame}

\begin{frame}[label=modelling-assumptions]{What did we assume when we wrote that?}
\begin{itemize}
\setlength{\itemsep}{.85em}
\item<1-> \key{Price-taking:} prices are given rather than negotiated.
\item<2-> \key{Linear prices:} every unit has the same unit price.
\item<3-> \key{Divisibility:} quantities can vary continuously.
\item<4-> \key{Decision horizon:} include saving or future consumption when relevant.
\end{itemize}
\uncover<5->{\question{What changes for a subscription, a quantity discount or an indivisible purchase?}}
\end{frame}

\begin{frame}[label=budget-geometry]{Currency invariance of the budget}
\uncover<1->{\key{Claim.} For every $t>0$,\[B(tp,tm)=B(p,m).\]}\par
\uncover<2->{\key{Proof.} For any $x\ge0$,}\par
\uncover<3->{\[tp\cdot x\le tm\quad\Longleftrightarrow\quad p\cdot x\le m.\]}\par
\uncover<4->{The same bundles are affordable. Only the currency units have changed.}\par
\end{frame}

\begin{frame}[label=proof-budget]{The budget is nonempty and compact}
\uncover<1->{\key{Claim.} If $p\gg0$ and $m>0$, then $B(p,m)$ is nonempty and compact.}\par
\uncover<2->{\key{Proof: nonempty.} The zero bundle is affordable.}\par
\uncover<3->{\key{Closed.} The inequalities $x_i\ge0$ and $p\cdot x\le m$ define closed sets.}\par
\uncover<4->{\key{Bounded.} Every affordable bundle satisfies $0\le x_i\le m/p_i$ for every $i$.}\par
\uncover<5->{A closed and bounded subset of finite-dimensional Euclidean space is compact.}\par
\end{frame}

\begin{frame}[label=existence]{Does an optimum exist?}
\uncover<1->{\key{Assumptions:} $u$ is continuous on $\R_+^L$, $p\gg0$ and $m>0$.}\par
\uncover<2->{The budget is nonempty and compact.}\par
\uncover<3->{By the extreme value theorem, a continuous function attains a maximum on it.}\par
\uncover<4->{Existence does not yet imply uniqueness, an interior solution or differentiability of demand.}\par
\end{frame}

\begin{frame}[label=existence-warning]{Which assumption is doing the work?}
\uncover<1->{Suppose $p_1=0$, $p_2=1$, $m=1$, and $u(x)=x_1$.}\par
\uncover<2->{Every quantity of good 1 is affordable. Utility is unbounded above, so there is no maximiser.}\par
\uncover<3->{Zero prices do not always destroy existence, but compactness is no longer guaranteed.}\par
\uncover<4->{Can you give a case with multiple maximisers even though the budget is compact?}\par
\uncover<5->{For $u=x_1+x_2$, $p=(1,1)$ and $m=1$, every bundle on $x_1+x_2=1$ is optimal.}\par
\end{frame}

\begin{frame}[label=marshallian]{Marshallian demand is initially a correspondence}
\uncover<1->{\[D(p,m)=\operatorname*{arg\,max}_{x\ge0,\,p\cdot x\le m}u(x).\]}\par
\uncover<2->{$D(p,m)$ contains \key{all} optimal bundles. It may contain more than one.}\par
\uncover<3->{When it is a singleton, write its unique element as $x(p,m)$: a demand function.}\par
\uncover<4->{It is the earlier choice rule applied specifically to budget menus: $D(p,m)=C(B(p,m))$.}\par
\end{frame}

\begin{frame}[label=homogeneity]{Proof: demand is homogeneous of degree zero}
\uncover<1->{Claim: $D(tp,tm)=D(p,m)$ for every $t>0$.}\par
\uncover<2->{$tp\cdot x\le tm\iff p\cdot x\le m$: divide by the positive number $t$.}\par
\uncover<3->{Therefore $B(tp,tm)=B(p,m)$. The objective $u$ is the same.}\par
\uncover<4->{The full argmax sets coincide. Neither uniqueness nor differentiability is needed.}\par
\uncover<5->{Example: $(p,m)=((2,1),12)$ and $((200,100),1200)$.}\par
\end{frame}

\begin{frame}[label=lagrangian-intuition]{Why introduce a Lagrangian?}
\uncover<1->{Write every inequality as a nonnegative slack: $m-p\cdot x\ge0$ and $x_i\ge0$.}\par
\uncover<2->{\[\mathcal L(x,\lambda,\mu)=u(x)+\lambda(m-p\cdot x)+\mu\cdot x,\qquad\lambda,\mu\ge0.\]}\par
\uncover<3->{A violated constraint lets its multiplier drive $\mathcal L$ down. At a feasible bundle the added terms are nonnegative.}\par
\uncover<4->{For the saddle argument, vary $x$ in an open convex domain $U$ where $u$ is defined.}\par
\end{frame}

\begin{frame}[label=saddle-definition]{What is a saddle point?}
\uncover<1->{A triple $(x^*,\lambda^*,\mu^*)$ is a global saddle if, for every $x\in U$ and $\lambda,\mu\ge0$,}\par
\uncover<2->{\[\mathcal L(x,\lambda^*,\mu^*)\le\mathcal L(x^*,\lambda^*,\mu^*)\le\mathcal L(x^*,\lambda,\mu).\]}\par
\uncover<3->{With multipliers fixed, $x^*$ maximises the Lagrangian.}\par
\uncover<4->{With quantities fixed, $(\lambda^*,\mu^*)$ minimises it over nonnegative multipliers.}\par
\end{frame}

\begin{frame}[label=theorem-saddle-optimum]{Theorem 1: a saddle gives a consumer optimum}
\uncover<1->{If $(x^*,\lambda^*,\mu^*)$ is a global saddle, then $x^*$ maximises $u$ over $x\in U$, $x\ge0$, $p\cdot x\le m$. We show:}\par
\begin{enumerate}
\setlength{\itemsep}{.45em}
\item<2-> \key{Feasibility:} $x^*\ge0$, $p\cdot x^*\le m$.
\item<3-> \key{Slackness:} $\lambda^*(m-p\cdot x^*)=0$, $\mu_i^*x_i^*=0$.
\item<4-> \key{Value:} $\mathcal L(x^*,\lambda^*,\mu^*)=u(x^*)$.
\item<5-> \key{Optimality:} every feasible $y$ satisfies $u(y)\le u(x^*)$.
\end{enumerate}
\end{frame}

\begin{frame}[label=saddle-proof-feasible]{Proof 1: feasibility}
\uncover<1->{At the saddle, nonnegative multipliers minimise $\mathcal L(x^*,\lambda,\mu)$.}\par
\uncover<2->{If $p\cdot x^*>m$, increasing $\lambda$ sends $\mathcal L$ to $-\infty$: impossible.}\par
\uncover<3->{If some $x_i^*<0$, increasing $\mu_i$ sends $\mathcal L$ to $-\infty$: also impossible.}\par
\uncover<4->{Therefore $p\cdot x^*\le m$ and $x^*\ge0$.}\par
\end{frame}

\begin{frame}[label=saddle-proof-slack]{Proof 1: slackness and equality of values}
\uncover<1->{If $\lambda^*>0$ and $m-p\cdot x^*>0$, reducing $\lambda$ to zero lowers $\mathcal L$: a contradiction.}\par
\uncover<2->{Likewise $\mu_i^*>0$ and $x_i^*>0$ cannot hold together.}\par
\uncover<3->{Thus every multiplier--slack product is zero:}\par
\uncover<4->{\[\mathcal L(x^*,\lambda^*,\mu^*)=u(x^*)+\underbrace{\lambda^*(m-p\cdot x^*)}_{0}+\underbrace{\mu^*\cdot x^*}_{0}=u(x^*).\]}\par
\end{frame}

\begin{frame}[label=saddle-proof-optimal]{Proof 1: compare with any affordable bundle}
\uncover<1->{For every feasible $y\in U$, $\lambda^*,\mu^*\ge0$ give $u(y)\le\mathcal L(y,\lambda^*,\mu^*)$.}\par
\uncover<2->{The saddle inequality then gives}\par
\uncover<3->{\[u(y)\le\mathcal L(y,\lambda^*,\mu^*)\le\mathcal L(x^*,\lambda^*,\mu^*)=u(x^*).\]}\par
\uncover<4->{This proves optimality. This direction needs no concavity or differentiability.}\par
\end{frame}

\begin{frame}[label=theorem-optimum-saddle]{Theorem 2: an optimum yields a saddle}
\uncover<1->{Let $p\gg0$, $m>0$, and let $u$ be differentiable and concave on $U$. For every consumer optimum $x^*$, suitable nonnegative multipliers give a global saddle. We show:}\par
\begin{enumerate}
\setlength{\itemsep}{.45em}
\item<2-> A common, nonnegative budget multiplier $\lambda^*$.
\item<3-> Nonnegative quantity-bound multipliers $\mu_i^*$.
\item<4-> Stationarity and complementary slackness.
\item<5-> Maximisation in $x$ and minimisation in the multipliers.
\end{enumerate}
\end{frame}

\begin{frame}[label=construct-saddle-multipliers]{Proof 2: construct the multipliers}
\uncover<1->{First take $x^*\ne0$. Choose a consumed good $j$, with $x_j^*>0$.}\par
\uncover<2->{\[\lambda^*:=\frac{u_j(x^*)}{p_j},\qquad\mu_i^*:=\begin{cases}0,&x_i^*>0,\\\lambda^*p_i-u_i(x^*),&x_i^*=0.\end{cases}\]}\par
\uncover<3->{Here $u_i=\partial u/\partial x_i$. The zero-bundle case will be handled separately.}\par
\end{frame}

\begin{frame}[label=saddle-lambda-proof]{Proof 2: a common nonnegative spending return}
\uncover<1->{If $\lambda^*<0$, then $u_j(x^*)<0$: slightly reducing consumed good $j$ improves utility and remains affordable. Hence $\lambda^*\ge0$.}\par
\uncover<2->{If two consumed goods had different $u_i/p_i$, transfer a small amount of spending toward the larger return.}\par
\uncover<3->{The transfer is feasible in either direction and has positive utility derivative, contradicting optimality.}\par
\uncover<4->{Therefore $u_i(x^*)/p_i=\lambda^*$ for every consumed good, independently of which $j$ was selected.}\par
\end{frame}

\begin{frame}[label=mu-sign-algebra]{Proof 2: why a negative bound multiplier fails}
\uncover<1->{Only $x_i^*=0$ needs checking: for a consumed good we already set $\mu_i^*=0$.}\par
\uncover<2->{Suppose $\mu_i^*<0$. By its definition,\[\lambda^*p_i-u_i(x^*)<0\quad\Rightarrow\quad\lambda^*<\frac{u_i(x^*)}{p_i}.\]}\par
\uncover<3->{For the consumed good $j$, $\lambda^*=u_j(x^*)/p_j$. Therefore\[\frac{u_i(x^*)}{p_i}>\frac{u_j(x^*)}{p_j}.\]}\par
\uncover<4->{The unconsumed good would give more marginal utility per unit of spending.}\par
\end{frame}

\begin{frame}[label=mu-sign-trade]{Proof 2: the profitable spending transfer}
\uncover<1->{Shift spending $\varepsilon$ from consumed good $j$ to unconsumed good $i$:}\par
\uncover<2->{\[x_i(\varepsilon)=\frac{\varepsilon}{p_i},\qquad x_j(\varepsilon)=x_j^*-\frac{\varepsilon}{p_j}.\]}\par
\uncover<3->{Keep other quantities fixed. For $0<\varepsilon<p_jx_j^*$, quantities remain nonnegative and total spending is unchanged.}\par
\uncover<4->{\[\left.\frac{d}{d\varepsilon}u(x(\varepsilon))\right|_{0+}=\frac{u_i(x^*)}{p_i}-\frac{u_j(x^*)}{p_j}=-\frac{\mu_i^*}{p_i}>0.\]}\par
\uncover<5->{A sufficiently small transfer strictly improves utility: contradiction. Thus $\mu_i^*\ge0$, with equality allowed.}\par
\end{frame}

\begin{frame}[label=proof-constructed-slackness]{Proof 2: stationarity and complementary slackness}
\uncover<1->{If $p\cdot x^*=m$, then $\lambda^*(m-p\cdot x^*)=0$.}\par
\uncover<2->{If the budget is slack and $x_j^*>0$, both a small increase and a small decrease in $x_j$ are feasible.}\par
\uncover<3->{Local optimality forces $u_j(x^*)=0$, hence $\lambda^*=u_j(x^*)/p_j=0$. The budget product is again zero.}\par
\uncover<4->{For every $i$, the construction gives $\mu_i^*x_i^*=0$: either the quantity or its multiplier is zero.}\par
\uncover<5->{The construction also gives $u_i(x^*)-\lambda^*p_i+\mu_i^*=0$.}\par
\end{frame}

\begin{frame}[label=saddle-max-proof]{Proof 2: maximise in quantities}
\uncover<1->{Fix the constructed multipliers and put $f(x)=\mathcal L(x,\lambda^*,\mu^*)$.}\par
\uncover<2->{Concave $u$ plus affine terms gives concave $f$ on $U$. Stationarity gives $\nabla f(x^*)=0$.}\par
\uncover<3->{A differentiable concave function lies below its tangent plane:}\par
\uncover<4->{\[f(y)\le f(x^*)+\nabla f(x^*)\cdot(y-x^*)=f(x^*)\qquad(y\in U).\]}\par
\end{frame}

\begin{frame}[label=proof-zero-bundle]{Proof 2: if the optimal bundle is zero}
\uncover<1->{If $x^*=0\in U$, a small increase in any good is affordable since $p\gg0$, $m>0$ and $U$ is open.}\par
\uncover<2->{Optimality implies $u_i(0)\le0$ for every $i$.}\par
\uncover<3->{Choose $\lambda^*=0$, $\mu_i^*=-u_i(0)\ge0$. Stationarity and complementary slackness hold.}\par
\uncover<4->{The same concavity argument makes $x^*=0$ maximise the fixed-multiplier Lagrangian.}\par
\end{frame}

\begin{frame}[label=saddle-min-proof]{Proof 2: minimise in multipliers and finish}
\uncover<1->{Feasibility and complementary slackness give $\mathcal L(x^*,\lambda^*,\mu^*)=u(x^*)$.}\par
\uncover<2->{For any $\lambda,\mu\ge0$, the nonnegative slacks give $\mathcal L(x^*,\lambda,\mu)\ge u(x^*)$.}\par
\uncover<3->{Combining this with maximisation in quantities,}\par
\uncover<4->{\[\mathcal L(x,\lambda^*,\mu^*)\le\mathcal L(x^*,\lambda^*,\mu^*)\le\mathcal L(x^*,\lambda,\mu).\]}\par
\uncover<5->{The constructed triple is a saddle. Both nonzero and zero optima are covered.}\par
\end{frame}

\begin{frame}[label=kkt]{The four Kuhn--Tucker groups}
\uncover<1->{\key{Stationarity:} $u_i(x^*)-\lambda^*p_i+\mu_i^*=0$.}\par
\uncover<2->{\key{Primal feasibility:} $p\cdot x^*\le m$, $x^*\ge0$.}\par
\uncover<3->{\key{Dual feasibility:} $\lambda^*\ge0$, $\mu^*\ge0$.}\par
\uncover<4->{\key{Complementary slackness:}
 \[\lambda^*(m-p\cdot x^*)=0,\qquad\mu_i^*x_i^*=0.\]}\par
\end{frame}

\begin{frame}[label=cd-interior]{Example 1: Cobb--Douglas}
\uncover<1->{\[u(x)=x_1^{\alpha}x_2^{1-\alpha},\qquad0<\alpha<1.\]}\par
\uncover<2->{\question{Could an optimum lie on an axis? Which representation would make the calculation easier?}}\par
\uncover<3->{An axis bundle gives zero utility; a strictly positive affordable bundle gives positive utility.}\par
\uncover<4->{Every optimum is therefore positive. We may now take logs and use an interior method.}\par
\end{frame}

\begin{frame}[label=cd-log]{Choose a convenient representation}
\uncover<1->{Maximising $u$ is equivalent, at the positive optimum, to maximising
 \[\widetilde u(x)=\alpha\log x_1+(1-\alpha)\log x_2.\]}\par
\uncover<2->{\[\mathcal L=\alpha\log x_1+(1-\alpha)\log x_2+\lambda(m-p_1x_1-p_2x_2).\]}\par
\uncover<3->{The log representation is concave on the positive domain.}\par
\uncover<4->{The bundle does not change, but the numerical utility and multiplier generally do.}\par
\end{frame}

\begin{frame}[label=cd-solve]{Solve the Cobb--Douglas conditions}
\uncover<1->{\[\frac{\alpha}{x_1}=\lambda p_1,\qquad\frac{1-\alpha}{x_2}=\lambda p_2.\]}\par
\uncover<2->{These imply $\lambda>0$, so slackness forces $p\cdot x=m$.}\par
\uncover<3->{Adding $p_1x_1=\alpha/\lambda$ and $p_2x_2=(1-\alpha)/\lambda$ gives $\lambda=1/m$.}\par
\uncover<4->{\[x_1=\frac{\alpha m}{p_1},\qquad x_2=\frac{(1-\alpha)m}{p_2}.\]}\par
\uncover<5->{At $\alpha=1/2$, $p=(2,1)$ and $m=12$, demand is $(3,6)$.}\par
\end{frame}

\begin{frame}[label=walras]{Walras’ law: an optimum exhausts the budget}
\uncover<1->{\key{Claim.} Under local nonsatiation, every existing optimum satisfies $p\cdot x^*=m$.}\par
\uncover<2->{\key{Proof.} Suppose slack is $\delta=m-p\cdot x^*>0$.}\par
\uncover<3->{If $y\ge0$ is within distance $\delta/(2\|p\|)$ of $x^*$, then $p\cdot y<m$.}\par
\uncover<4->{LNS supplies a strictly better such $y$, contradicting optimality.}\par
\end{frame}

\begin{frame}[label=substitutes]{Example 2: perfect substitutes}
\uncover<1->{\[u(x)=a x_1+b x_2,\qquad a,b>0.\]}\par
\uncover<2->{\question{Where would you spend the next unit of money? Can the optimum be nonunique?}}\par
\uncover<3->{Compare $a/p_1$ and $b/p_2$. Buy $(m/p_1,0)$ if the first is larger; buy $(0,m/p_2)$ if the second is larger.}\par
\uncover<4->{If the ratios are equal, every budget-line bundle is optimal.}\par
\end{frame}

\begin{frame}[label=corner-check]{Verify a corner with the full KKT system}
\uncover<1->{Let $u(x)=2x_1+x_2$, $p=(3,1)$ and $m=12$.}\par
\uncover<2->{The spending returns are $2/3$ and $1$, so $x^*=(0,12)$.}\par
\uncover<3->{Stationarity and slackness give $\lambda=1$, $\mu_1=1$, $\mu_2=0$.}\par
\uncover<4->{Check: $2-3\lambda+\mu_1=0$, $1-\lambda+\mu_2=0$.}\par
\uncover<5->{The zero quantity is optimal precisely because its marginal return is too low.}\par
\end{frame}

\begin{frame}[label=quasilinear]{Example 3: lessons and money left over}
\uncover<1->{\[u(a,q)=a+4\log(1+q),\qquad a+rq\le m,\quad a,q\ge0.\]}\par
\uncover<2->{The price of $a$ is 1; the lesson price is $r>0$; wealth is $m>0$.}\par
\uncover<3->{\question{Must both goods be consumed? How could we reduce this to one variable?}}\par
\uncover<4->{Budget exhaustion gives $a=m-rq$ and $0\le q\le m/r$.}\par
\end{frame}

\begin{frame}[label=quasilinear-solve]{Solve first, then respect the quantity bounds}
\uncover<1->{\[\max_{0\le q\le m/r}\;m-rq+4\log(1+q).\]}\par
\uncover<2->{The derivative $-r+4/(1+q)$ is strictly decreasing.}\par
\uncover<3->{The stationary quantity is $4/r-1$. Impose both bounds:}\par
\uncover<4->{\[q^*=\min\left\{\frac mr,\max\left\{\frac4r-1,0\right\}\right\},\quad a^*=m-rq^*.\]}\par
\end{frame}

\begin{frame}[label=quasilinear-regimes]{Three regimes, one decision problem}
\uncover<1->{If $r\ge4$: no tutoring, $q^*=0$, $a^*=m$.}\par
\uncover<2->{If $0<r<4$ and $m\le4-r$: the no-borrowing bound binds, $q^*=m/r$, $a^*=0$.}\par
\uncover<3->{If $0<r<4$ and $m>4-r$: both goods are consumed, $q^*=4/r-1$, $a^*=m-4+r$.}\par
\uncover<4->{At $r=1$: income 2 gives $(a,q)=(0,2)$; income 8 gives $(5,3)$.}\par
\uncover<5->{At $m=4-r$, both formulas agree; a binding bound may have zero multiplier.}\par
\end{frame}

\begin{frame}[label=complements]{Perfect complements: solve at the kink}
\uncover<1->{\[u(x)=\min\{x_1,x_2\}.\]}\par
\uncover<2->{\question{What proportions would you buy, and why would a surplus be wasteful?}}\par
\uncover<3->{Utility $t$ costs at least $(p_1+p_2)t$, with cheapest bundle $(t,t)$.}\par
\uncover<4->{\[x_1^*=x_2^*=\frac{m}{p_1+p_2};\qquad p=(2,1),\ m=12\ \Rightarrow\ (4,4).\]}\par
\uncover<5->{The target-cost argument works even though smooth tangency fails at the kink.}\par
\end{frame}

\begin{frame}[label=demand-convex]{Proof: convex preferences give a convex demand set}
\uncover<1->{Assume preferences are convex and take $x,y\in D(p,m)$.}\par
\uncover<2->{For $t\in[0,1]$, $z_t=tx+(1-t)y$ is affordable because $B(p,m)$ is convex.}\par
\uncover<3->{For every affordable $w$, optimality gives $x\succeq w$ and $y\succeq w$.}\par
\uncover<4->{Convex preferences imply $z_t\succeq w$ for every such $w$. Thus $z_t\in D(p,m)$.}\par
\uncover<5->{Hence $D(p,m)$ is convex. It can still be a whole line segment.}\par
\end{frame}

\begin{frame}[label=unique-demand]{Strictly convex preferences: uniqueness}
\uncover<1->{\key{Claim.} On a convex budget set, strictly convex preferences give at most one optimum.}\par
\uncover<2->{\key{Proof.} If distinct $x,y$ were optimal, they would be indifferent.}\par
\uncover<3->{Their midpoint is affordable and, by strict convexity, strictly preferred: a contradiction.}\par
\uncover<4->{With existence already established, there is a unique demanded bundle.}\par
\end{frame}

\begin{frame}[label=indirect-utility]{Indirect utility: the optimal value}
\uncover<1->{\[v(p,m)=\max_{x\ge0,\,p\cdot x\le m}u(x).\]}\par
\uncover<2->{Demand tells us \key{which bundle}; indirect utility tells us \key{the attained value}.}\par
\uncover<3->{Even if $D(p,m)$ contains several bundles, every one gives the same value $v(p,m)$.}\par
\uncover<4->{It is a function of the environment, not of a chosen bundle.}\par
\end{frame}

\begin{frame}[label=v-cd]{Compute the value: Cobb--Douglas}
\uncover<1->{For $\widetilde u=\alpha\log x_1+(1-\alpha)\log x_2$, substitute the optimal quantities:}\par
\uncover<2->{\[v_{\log}=\alpha\log\frac{\alpha m}{p_1}+(1-\alpha)\log\frac{(1-\alpha)m}{p_2}.\]}\par
\uncover<3->{Thus $v_{\log}=\log m-\alpha\log p_1-(1-\alpha)\log p_2+c_\alpha$, where $c_\alpha=\alpha\log\alpha+(1-\alpha)\log(1-\alpha)$.}\par
\uncover<4->{For the original product index, $v_{\mathrm{raw}}=\exp(v_{\log})$. State which scale you use.}\par
\end{frame}

\begin{frame}[label=v-other-models]{Compute the value: substitutes and complements}
\uncover<1->{For $u=ax_1+bx_2$, spend on the higher-return good:}\par
\uncover<2->{\[v(p,m)=m\max\left\{\frac a{p_1},\frac b{p_2}\right\}.\]}\par
\uncover<3->{For $u=\min\{x_1,x_2\}$, use the equal-quantity optimum:}\par
\uncover<4->{\[v(p,m)=\frac{m}{p_1+p_2}.\]}\par
\end{frame}

\begin{frame}[label=roy-quasilinear-value]{Compute the value: the logarithmic model}
\uncover<1->{Return to $u=a+4\log(1+q)$ and $a+rq\le m$. Substitute the demands:}\par
\uncover<2->{\[v(r,m)=\begin{cases}m,&r\ge4,\\4\log(1+m/r),&0<r<4,\ m\le4-r,\\m-4+r+4\log(4/r),&0<r<4,\ m>4-r.\end{cases}\]}\par
\uncover<3->{\question{What happens to this value when wealth increases or the lesson price rises?}}\par
\end{frame}

\begin{frame}[label=v-properties]{Indirect utility: the properties to prove}
\begin{itemize}
\setlength{\itemsep}{.85em}
\item<1-> Maintain continuous $u$, $p\gg0$ and $m>0$, so optimal values exist.
\item<2-> $v(tp,tm)=v(p,m)$; higher prices cannot increase $v$.
\item<3-> More wealth weakly increases $v$; under LNS the increase is strict.
\item<4-> $v$ is continuous on the positive-price, positive-wealth domain.
\end{itemize}
\end{frame}

\begin{frame}[label=v-quasiconvex]{A less obvious property: quasiconvexity}
\uncover<1->{For $0\le t\le1$, set $(p^t,m^t)=t(p,m)+(1-t)(q,n)$.}\par
\uncover<2->{\[v(p^t,m^t)\le\max\{v(p,m),v(q,n)\}.\]}\par
\uncover<3->{If a bundle is unaffordable at both endpoints, it is unaffordable at their mixture too.}\par
\uncover<4->{So the mixed budget cannot deliver utility above both endpoint values.}\par
\end{frame}

\begin{frame}[label=envelope-general-problem]{A general parameterised optimisation problem}
\uncover<1->{\[V(\theta)=\max_x f(x,\theta)\quad\text{subject to}\quad g_j(x,\theta)\ge0,\quad h_k(x,\theta)=0.\]}\par
\uncover<2->{The parameter varies in an open set; objective and constraints are continuously differentiable.}\par
\uncover<3->{The ambient decision domain is open; all quantity bounds are included among the constraints.}\par
\uncover<4->{Use $\lambda_j\ge0$ for inequalities and unrestricted $\nu_k$ for equalities.}\par
\end{frame}

\begin{frame}[label=envelope-general-lagrangian]{The general Lagrangian}
\uncover<1->{\[\mathcal L=f(x,\theta)+\sum_j\lambda_jg_j(x,\theta)+\sum_k\nu_kh_k(x,\theta).\]}\par
\uncover<2->{At an optimum: $\mathcal L_x=0$, $g_j\ge0$, $h_k=0$, $\lambda_j\ge0$ and $\lambda_jg_j=0$.}\par
\uncover<3->{Assume a continuously differentiable selection of optimal choices and KKT multipliers exists near the parameter under consideration.}\par
\uncover<4->{This is a sufficient smooth-branch regularity assumption, to be checked in an application.}\par
\end{frame}

\begin{frame}[label=envelope-theorem]{Envelope theorem: a constrained smooth version}
\uncover<1->{Under the preceding smooth-branch assumptions,}\par
\uncover<2->{\[\frac{\partial V}{\partial\theta_\ell}=\frac{\partial\mathcal L}{\partial\theta_\ell}(x^*,\lambda^*,\nu^*;\theta).\]}\par
\uncover<3->{On the right, hold decisions and multipliers fixed when taking the partial derivative.}\par
\uncover<4->{Thus $V_{\theta_\ell}=f_{\theta_\ell}+\sum_j\lambda_j^*(g_j)_{\theta_\ell}+\sum_k\nu_k^*(h_k)_{\theta_\ell}$, evaluated at the optimum.}\par
\end{frame}

\begin{frame}[label=envelope-proof-chain]{Proof: differentiate along an optimal branch}
\uncover<1->{Feasibility and slackness give $V(\theta)=\mathcal L(x^*(\theta),\lambda^*(\theta),\nu^*(\theta);\theta)$.}\par
\uncover<2->{Fix one coordinate $t$ of $\theta$. The chain rule gives}\par
\uncover<3->{\[\frac{dV}{dt}=\mathcal L_t+\mathcal L_x x_t^*+\sum_jg_j(\lambda_j^*)_t+\sum_kh_k(\nu_k^*)_t.\]}\par
\uncover<4->{Stationarity kills the $\mathcal L_x$ term; equality feasibility kills the $h_k$ terms.}\par
\end{frame}

\begin{frame}[label=envelope-proof-slackness]{Proof: why multiplier derivatives disappear}
\uncover<1->{For each inequality, examine $g_j(\lambda_j^*)_t$.}\par
\uncover<2->{If $\lambda_j^*>0$, complementary slackness gives $g_j=0$.}\par
\uncover<3->{If $\lambda_j^*=0$, its differentiable nonnegative branch has a local minimum at zero, so $(\lambda_j^*)_t=0$.}\par
\uncover<4->{Every extra term is zero. Hence\[\frac{dV}{dt}=\mathcal L_t.\]}\par
\end{frame}

\begin{frame}[label=envelope-formulas]{Income and price derivatives of indirect utility}
\uncover<1->{The parameter-dependent constraint is $m-p\cdot x\ge0$. Quantity bounds do not depend on prices or wealth.}\par
\uncover<2->{\[v_m=\mathcal L_m=\lambda^*.\]}\par
\uncover<3->{\[v_{p_i}=\mathcal L_{p_i}=-\lambda^*x_i^*.\]}\par
\uncover<4->{Where the envelope assumptions hold, $\lambda^*$ is the marginal value of wealth in the chosen utility units.}\par
\end{frame}

\begin{frame}[label=roy]{Roy’s identity: recover demand from value}
\uncover<1->{Assume continuous utility, LNS, $p\gg0$, $m>0$, differentiable $v$ and $v_m>0$.}\par
\uncover<2->{Every optimiser satisfies\[x_i^*=-\frac{v_{p_i}(p,m)}{v_m(p,m)}.\]}\par
\uncover<3->{At a smooth regular optimum, the envelope formulas give the ratio by cancelling $\lambda^*$.}\par
\uncover<4->{The next proof needs no differentiable demand or differentiable direct utility.}\par
\end{frame}

\begin{frame}[label=proof-roy]{Roy without differentiating demand}
\uncover<1->{Let $x^*\in D(p,m)$. LNS gives $p\cdot x^*=m$.}\par
\uncover<2->{At any nearby price vector $q$, the bundle $x^*$ is affordable with wealth $q\cdot x^*$. Thus $F(q)=v(q,q\cdot x^*)\ge u(x^*)=F(p)$.}\par
\uncover<3->{If $v$ is differentiable, $F$ has a minimum at $p$:
 \[0=F_{q_i}(p)=v_{p_i}(p,m)+v_m(p,m)x_i^*.\]}\par
\uncover<4->{If $v_m>0$, every optimiser satisfies the same Roy formula, so demand is unique.\quad}\par
\vfill\navto{roy}{Back}
\end{frame}

\begin{frame}[label=roy-example]{Check Roy in the Cobb--Douglas example}
\uncover<1->{\[v=\log m-\alpha\log p_1-(1-\alpha)\log p_2+c_\alpha.\]}\par
\uncover<2->{\[v_m=1/m,\qquad v_{p_1}=-\alpha/p_1,\qquad v_{p_2}=-(1-\alpha)/p_2.\]}\par
\uncover<3->{\[x_1=-\frac{-\alpha/p_1}{1/m}=\frac{\alpha m}{p_1}.\]}\par
\uncover<4->{Likewise $x_2=(1-\alpha)m/p_2$. The demand agrees with the direct solution.}\par
\end{frame}

\begin{frame}[label=roy-subs-value]{Roy example 2: substitutes and a corner}
\uncover<1->{For $u=ax_1+bx_2$,\[v(p,m)=m\max\left\{\frac a{p_1},\frac b{p_2}\right\}.\]}\par
\uncover<2->{\question{Differentiate in a region where good 1 gives the strictly larger spending return.}}\par
\uncover<3->{If $a/p_1>b/p_2$, then $v=am/p_1$, $v_m=a/p_1$, $v_{p_1}=-am/p_1^2$ and $v_{p_2}=0$.}\par
\uncover<4->{Roy gives $x_1=m/p_1$, $x_2=0$.}\par
\end{frame}

\begin{frame}[label=roy-complements]{Roy example 4: a kink in direct utility}
\uncover<1->{For $u=\min\{x_1,x_2\}$,\[v(p,m)=\frac{m}{p_1+p_2}.\]}\par
\uncover<2->{$v_m=1/(p_1+p_2)$ and $v_{p_i}=-m/(p_1+p_2)^2$.}\par
\uncover<3->{\[x_i=-\frac{v_{p_i}}{v_m}=\frac{m}{p_1+p_2}\qquad(i=1,2).\]}\par
\uncover<4->{Direct utility has a kink, but indirect utility is differentiable. The direct proof of Roy still applies.}\par
\end{frame}

\uncover<1->{For the logarithmic value computed earlier, use $q^*=-v_r/v_m$.}\par

\begin{frame}[label=preview-two-problems]{Next block: two ways to ask the consumer question}
\uncover<1->{\key{Utility maximisation:} given $p,m$, how much utility can the consumer attain?}\par
\uncover<2->{\key{Expenditure minimisation:} given $p,\bar u$, how little spending attains the target?}\par
\uncover<3->{One fixes resources; the other fixes a level of the same preferences.}\par
\end{frame}

\begin{frame}[label=preview-emp]{Expenditure minimisation}
\uncover<1->{\[e(p,\bar u)=\min_{x\ge0,\,u(x)\ge\bar u}p\cdot x.\]}\par
\uncover<2->{\[H(p,\bar u)=\operatorname*{arg\,min}_{x\ge0,\,u(x)\ge\bar u}p\cdot x.\]}\par
\uncover<3->{$e$ is the expenditure function; $H$ is the Hicksian demand correspondence.}\par
\uncover<4->{Assume the target is attainable. A singleton is denoted $h(p,\bar u)$.}\par
\end{frame}

\begin{frame}[label=preview-existence]{Why an unbounded feasible set can still have a minimum}
\uncover<1->{Assume $p\gg0$, continuous $u$, and pick a bundle $\widehat x$ with $u(\widehat x)\ge\bar u$.}\par
\uncover<2->{Only consider target-feasible bundles with $p\cdot x\le p\cdot\widehat x$.}\par
\uncover<3->{This set is nonempty, closed and bounded, hence compact.}\par
\uncover<4->{A continuous cost attains a minimum there; excluded bundles are more expensive.}\par
\end{frame}

\begin{frame}[label=preview-e-properties]{Expenditure: a few properties for next time}
\uncover<1->{$e(tp,\bar u)=t e(p,\bar u)$ for $t>0$; raising prices componentwise cannot lower expenditure.}\par
\uncover<2->{For a fixed attainable target, expenditure is concave in prices:}\par
\uncover<3->{\[e(tp+(1-t)q,\bar u)\ge t e(p,\bar u)+(1-t)e(q,\bar u).\]}\par
\uncover<4->{It is the minimum over target-feasible bundles of linear price functions.}\par
\end{frame}

\begin{frame}[label=preview-duality]{Two identities to develop next time}
\uncover<1->{Under continuous LNS preferences, positive prices and an attainable target $\bar u>u(0)$:}\par
\uncover<2->{\[e\bigl(p,v(p,m)\bigr)=m,\qquad v\bigl(p,e(p,\bar u)\bigr)=\bar u.\]}\par
\uncover<3->{For unique demands,
 \[h\bigl(p,v(p,m)\bigr)=x(p,m).\]}\par
\uncover<4->{These connect the two problems; they are not an assumption that the consumer literally solves both.}\par
\end{frame}
